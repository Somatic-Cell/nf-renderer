cmake_minimum_required(VERSION 3.18)
project(M4RainValidation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(m4_rain_test_asset STATIC "${ROOT}/src/m4_nf/asset.cpp")
target_include_directories(m4_rain_test_asset PUBLIC "${ROOT}/include")
add_executable(m4_rain_tests test_rain.cpp)
target_link_libraries(m4_rain_tests PRIVATE m4_rain_test_asset)
add_executable(m4_nf_previous_tests "${ROOT}/tests/m4_nf/test_core.cpp")
target_link_libraries(m4_nf_previous_tests PRIVATE m4_rain_test_asset)
foreach(t m4_rain_test_asset m4_rain_tests m4_nf_previous_tests)
  if(MSVC)
    target_compile_options(${t} PRIVATE /utf-8 /fp:precise)
    target_compile_definitions(${t} PRIVATE NOMINMAX)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
  endif()
endforeach()
enable_testing()
add_test(NAME rain_core COMMAND m4_rain_tests "${ROOT}/assets/m4_nf/best/model.bin")
add_test(NAME previous_nf_core COMMAND m4_nf_previous_tests "${ROOT}/assets/m4_nf/best/model.bin")
# Optional full Scene JSON regression: uses the repository's real dependency.
find_path(M4_JSON_INCLUDE nlohmann/json.hpp HINTS "${ROOT}/ext/json/include" "${ROOT}/ext/json/single_include")
if(M4_JSON_INCLUDE)
  add_executable(m4_rain_scene_tests test_scene.cpp)
  target_include_directories(m4_rain_scene_tests PRIVATE "${ROOT}/include" "${M4_JSON_INCLUDE}")
  if(MSVC)
    target_compile_options(m4_rain_scene_tests PRIVATE /utf-8 /fp:precise)
    target_compile_definitions(m4_rain_scene_tests PRIVATE NOMINMAX)
  endif()
  add_test(NAME rain_scene_json COMMAND m4_rain_scene_tests "${ROOT}")
else()
  message(STATUS "nlohmann/json.hpp unavailable: JSON compile/runtime regression NOT RUN")
endif()
