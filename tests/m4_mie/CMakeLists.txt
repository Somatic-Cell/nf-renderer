cmake_minimum_required(VERSION 3.18)
project(M4MieReferenceTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(m4_mie_real_lut_test test_real_lut.cpp "${REPO}/src/m4_mie/asset.cpp")
target_include_directories(m4_mie_real_lut_test PRIVATE "${REPO}/include")
if(MSVC)
  target_compile_definitions(m4_mie_real_lut_test PRIVATE NOMINMAX)
  target_compile_options(m4_mie_real_lut_test PRIVATE /utf-8 /fp:precise)
else()
  target_compile_options(m4_mie_real_lut_test PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
enable_testing()
add_test(NAME m4_mie_real_lut COMMAND m4_mie_real_lut_test "${REPO}/assets/m4_mie/mie_sphere_1mm.bin")
